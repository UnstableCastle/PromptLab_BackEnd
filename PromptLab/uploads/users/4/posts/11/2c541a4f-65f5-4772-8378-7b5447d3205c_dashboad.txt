<script>
function updateClock() {
    const now = new Date();

    const options = {
        weekday: "long",
        year: "numeric",
        month: "long",
        day: "numeric"
    };

    const date = now.toLocaleDateString(undefined, options);
    const time = now.toLocaleTimeString();

    document.getElementById("clock").innerHTML =
        date + " | " + time;
}

updateClock();
setInterval(updateClock, 1000);
</script>
